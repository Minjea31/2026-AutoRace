cmake_minimum_required(VERSION 3.16)
project(autorace_lane_deep_lite)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs std_msgs cv_bridge)
  find_package(${dep} REQUIRED)
endforeach()
find_package(OpenCV REQUIRED)
find_package(yaml-cpp REQUIRED)

add_library(lane_deep_lite_core STATIC src/deep_lane.cpp)
set_target_properties(lane_deep_lite_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(lane_deep_lite_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lane_deep_lite_core PUBLIC ${OpenCV_LIBS} yaml-cpp)

add_executable(lane_deep_lite_node src/node.cpp)
target_link_libraries(lane_deep_lite_node lane_deep_lite_core)
ament_target_dependencies(lane_deep_lite_node rclcpp sensor_msgs std_msgs)
install(TARGETS lane_deep_lite_node DESTINATION lib/${PROJECT_NAME})

# Bag evaluator: only where rosbag2 is installed.
find_package(rosbag2_cpp QUIET)
if(rosbag2_cpp_FOUND)
  add_executable(lane_deep_lite_eval src/eval.cpp)
  target_link_libraries(lane_deep_lite_eval lane_deep_lite_core)
  ament_target_dependencies(lane_deep_lite_eval rclcpp sensor_msgs cv_bridge rosbag2_cpp)
  install(TARGETS lane_deep_lite_eval DESTINATION lib/${PROJECT_NAME})
endif()

# Qt tuner: only where Qt5 and rosbag2 are installed (the laptop).
find_package(Qt5 QUIET COMPONENTS Widgets)
find_package(ament_index_cpp QUIET)
if(Qt5_FOUND AND rosbag2_cpp_FOUND AND ament_index_cpp_FOUND)
  set(CMAKE_AUTOMOC OFF)
  add_executable(lane_deep_lite_tuner src/tuner.cpp)
  target_link_libraries(lane_deep_lite_tuner lane_deep_lite_core Qt5::Widgets)
  target_compile_definitions(lane_deep_lite_tuner PRIVATE DEEP_LITE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  ament_target_dependencies(lane_deep_lite_tuner rclcpp sensor_msgs cv_bridge rosbag2_cpp ament_index_cpp)
  install(TARGETS lane_deep_lite_tuner DESTINATION lib/${PROJECT_NAME})
else()
  message(STATUS "Qt5/rosbag2 not found: lane_deep_lite_tuner skipped")
endif()

install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_deep_lane test/test_deep_lane.cpp)
  target_link_libraries(test_deep_lane lane_deep_lite_core)
  ament_add_gtest(test_fit_chain test/test_fit_chain.cpp)
  target_include_directories(test_fit_chain PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(test_fit_chain ${OpenCV_LIBS} yaml-cpp)
endif()
ament_package()
